package test

import (
	"bytes"
	"errors"
	"testing"

	"github.com/redraskal/r6-dissect/dissect"
)

// TestReader_Read_HeaderOnlyReplay: a Y8S4+ (chunked) round file cut before its first zstd
// frame has a full header and no replay data. Read must return ErrNoReplayData, not panic.
func TestReader_Read_HeaderOnlyReplay(t *testing.T) {
	b := readFixture(t, solidSnake)
	end := bytes.Index(b, []byte{0x28, 0xB5, 0x2F, 0xFD}) // first zstd frame
	if end <= 0 {
		t.Fatal("fixture has no zstd frame after its header")
	}
	r, err := dissect.NewReader(bytes.NewReader(b[:end]))
	if err != nil {
		t.Fatalf("NewReader(): %v", err)
	}
	if err := r.Read(); !errors.Is(err, dissect.ErrNoReplayData) {
		t.Fatalf("Read(): want ErrNoReplayData, got %v", err)
	}
}
